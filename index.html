Copy URL → ESP32 SCRIPT_URL + dashboard */ var SHEET_NAME = "Attendance"; function doGet(e) { try { var p = (e && e.parameter) ? e.parameter : {}; if (p.action === "read") return readSheet(); return writeRow(p); } catch(err) { return jsonOut({ result:"error", msg: err.message }); } } function doPost(e) { try { var p = {}; if (e && e.parameter) p = e.parameter; if (e && e.postData && e.postData.contents) { var pairs = e.postData.contents.split("&"); pairs.forEach(function(pair) { var kv = pair.split("="); if (kv.length === 2) p[kv[0]] = decodeURIComponent(kv[1].replace(/\+/g," ")); }); } return writeRow(p); } catch(err) { return jsonOut({ result:"error", msg: err.message }); } } function writeRow(p) { var uid = clean(p.uid) || "UNKNOWN"; var name = clean(p.name) || "UNKNOWN"; var grade = clean(p.grade) || "-"; var status = clean(p.status) || "UNKNOWN"; if (uid === "UNKNOWN" && name === "UNKNOWN") { return jsonOut({ result:"ignored" }); } Logger.log("WRITE uid=%s name=%s grade=%s status=%s", uid, name, grade, status); var sheet = getOrCreateSheet(); var ts = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"); sheet.appendRow([ts, uid, name, grade, status]); var row = sheet.getLastRow(); // Status colour (col 5) var sc = sheet.getRange(row, 5); var isAuth = (status === "Authorised"); sc.setBackground(isAuth ? "#d9ead3" : "#fce5cd"); sc.setFontColor (isAuth ? "#274e13" : "#b45309"); sc.setFontWeight("bold"); sc.setHorizontalAlignment("center"); Logger.log("Row %s OK", row); return jsonOut({ result:"success", row:row, ts:ts, received:{ uid:uid, name:name, grade:grade, status:status } }); } function readSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) return jsonOut({ data:[] }); var rows = sheet.getDataRange().getValues(); if (rows.length < 2) return jsonOut({ data:[] }); var keys = rows[0].map(function(h){ return String(h).toLowerCase().trim(); }); var data = rows.slice(1).map(function(row){ var obj = {}; keys.forEach(function(k,i){ var v = row[i]; obj[k] = (v instanceof Date) ? Utilities.formatDate(v, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss") : String(v); }); return obj; }); return jsonOut({ data:data }); } function getOrCreateSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); var h = sheet.getRange(1,1,1,5); h.setValues([["Timestamp","UID","Name","Grade","Status"]]); h.setFontWeight("bold"); h.setBackground("#1D9E75"); h.setFontColor("#FFFFFF"); h.setHorizontalAlignment("center"); sheet.setFrozenRows(1); sheet.setColumnWidth(1,175); sheet.setColumnWidth(2,120); sheet.setColumnWidth(3,130); sheet.setColumnWidth(4,100); sheet.setColumnWidth(5,115); } return sheet; } function clean(val) { if (val === null || val === undefined) return ""; var s = String(val).trim(); try { s = decodeURIComponent(s); } catch(e) {} return s.replace(/\+/g," ").trim(); } function jsonOut(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function setupSheet() { getOrCreateSheet(); Logger.log("Sheet ready: " + SHEET_NAME); } function testWrite() { Logger.log(writeRow({ uid:"73BA2325", name:"Rilwan", grade:"Grade-A", status:"Authorised" }).getContent()); } function testDenied() { Logger.log(writeRow({ uid:"AABBCCDD", name:"Unknown", grade:"-", status:"Denied" }).getContent()); } function testRead() { var r = JSON.parse(readSheet().getContent()); Logger.log("Rows: " + r.data.length); }
